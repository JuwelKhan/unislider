Usage

        
Three ways in — pick whichever suits your site.

      

      

        

          
1 · Markup only


<link rel="stylesheet" href="unislider.min.css">
<script src="unislider.min.js" defer></script>

<div data-unislider="rainy">
  <div data-us-bg="slide-1.jpg" data-us-rain="storm">
    <h2 data-us-layer="rainfall" data-us-split="chars"
        data-us-stagger="random">Hello</h2>
  </div>
  <div data-us-bg="slide-2.jpg" data-us-transition="ripple">…</div>
</div>

        

        

          
2 · JavaScript options


const slider = new UniSlider('#hero', {
  preset: 'hero',
  transition: ['rain', 'drip', 'cube'],
  rain: 'storm',         // live weather on a canvas
  ripple: true,          // water ripples on click
  mode: 'auto',          // auto | manual | hybrid
  autoplay: { delay: 6000 },
  breakpoints: { 768: { layout: 'fullscreen' } },
});

slider.on('change', ({ index }) => …);
slider.next();

        

        

          
3 · Templates


UniSlider.create('#reviews', {
  template: 'testimonials',
  slides: [
    { quote: 'Fantastic!', name: 'Ana', role: 'CEO' },
    { quote: 'So easy.',   name: 'Rui', role: 'Dev' },
  ],
});
